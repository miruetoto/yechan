연구 ▷ HST ▷ HST height 분해 — 마팅게일 잔차 이론치 검증

Author

리이

Published

May 22, 2026

HST height 는 drift + martingale 으로 분해된다 (에이 ABC증명 Theorem C Step 1):

\[h(v_i, t) = \bar{b}\sum_{s=1}^t \mathbf{1}\{X_s=v_i\} + M_t^{(i)}, \qquad M_t^{(i)} := \sum_{s=1}^t (b'_s - \bar{b})\mathbf{1}\{X_s=v_i\}\]

\(M_t^{(i)}\) 가 실제로 마팅게일인지, 그리고 이론 분산

\[\operatorname{Var}(M_t^{(i)}) \approx \frac{b^2}{12}\,\rho_i\,t \quad \Rightarrow \quad \frac{M_t^{(i)}}{\sqrt{t}} \xrightarrow{d} N\!\Bigl(0,\;\frac{b^2\rho_i}{12}\Bigr)\]

과 맞는지 세 가지 그래프 구조에서 확인한다. 공통 설정.

import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
plt.style.use("seaborn-v0_8-whitegrid")
plt.rcParams.update({
    "text.usetex": True,
    "text.latex.preamble": r"\usepackage{amsmath,amssymb}",
    "font.family": "serif",
    "font.size": 9,
    "axes.titlesize": 10,
})
rng = np.random.default_rng(20260522)

b = 1.0; b_bar = b / 2.0; var_b = b**2 / 12  # Var(Unif(0,b))
T = 8000; NPATH = 100

def simulate(degrees, T, NPATH):
    """degree-prop random walk + height decomposition"""
    n = len(degrees)
    prob = np.array(degrees, dtype=float) / sum(degrees)
    walker  = rng.choice(n, size=(NPATH, T), p=prob)
    b_prime = rng.uniform(0, b, size=(NPATH, T))
    rho_emp = np.array([walker[:, T//2:].mean() == i for i in range(n)])  # rough
    # per-node martingale and drift
    ind  = np.stack([(walker == i).astype(float) for i in range(n)], axis=0)  # (n, NPATH, T)
    M    = np.cumsum((b_prime[None] - b_bar) * ind, axis=2)   # (n, NPATH, T)
    h    = np.cumsum(b_prime[None] * ind, axis=2)              # (n, NPATH, T)
    rho  = ind.mean(axis=(1, 2))                               # (n,) long-run freq
    return h, M, rho, prob
# 세 그래프 정의
graphs = {
    r"Star $K_{1,5}$":   [5, 1, 1, 1, 1, 1],          # hub, 5 leaves
    r"Wheel $W_{1,5}$":  [5, 3, 3, 3, 3, 3],           # hub, 5 rim (rim deg=3)
    r"Balanced $C_6$":   [2, 2, 2, 2, 2, 2],           # 6-cycle
}
node_labels = {
    r"Star $K_{1,5}$":  [(0, "hub"), (1, "leaf")],
    r"Wheel $W_{1,5}$": [(0, "hub"), (1, "rim")],
    r"Balanced $C_6$":  [(0, r"$v_0$"), (1, r"$v_1$")],
}

results = {}
for name, deg in graphs.items():
    h, M, rho, prob_theory = simulate(deg, T, NPATH)
    results[name] = dict(h=h, M=M, rho=rho, prob=prob_theory)
    print(f"\n{name}")
    for i, lbl in node_labels[name]:
        sig_theory = np.sqrt(var_b * prob_theory[i])
        sig_emp    = (M[i, :, -1] / np.sqrt(T)).std()
        print(f"  node {lbl}: rho_emp={rho[i]:.3f}  rho_theory={prob_theory[i]:.3f} | "
              f"sigma_theory={sig_theory:.4f}  sigma_emp={sig_emp:.4f}")

각 그래프, 각 노드에서 \(\rho_i\) 경험값 ≈ 이론값, \(\sigma = \sqrt{b^2\rho_i/12}\) 도 일치하는지 확인.

t_arr = np.arange(1, T + 1)
fig, axes = plt.subplots(len(graphs), 4, figsize=(8, 7))

        rho_th = prob[node_idx]

        ax0 = axes[row, col_offset * 2]      # path
        ax1 = axes[row, col_offset * 2 + 1]  # M_t/t → 0  or  distribution

        # --- path plot ---

            ax0.plot(t_arr, M_node[k], lw=0.4, alpha=0.3, color=f"C{col_offset}")

        ax0.axhline(0, color="r", lw=0.8, ls="--")
        ax0.set_title(f"{name}: {node_lbl}", fontsize=8)
        ax0.set_xlabel(r"$t$"); ax0.set_ylabel(r"$M_t$")

        # --- M_t/t → 0 AND distribution ---

        ax1.fill_between(t_arr,
                         ratio.mean(0) - ratio.std(0),
                         ratio.mean(0) + ratio.std(0),
                         alpha=0.25, color=f"C{col_offset}")
        ax1.axhline(0, color="r", lw=0.8, ls="--")
        # annotate sigma_theory
        ax1.text(0.98, 0.95,
                 rf"$\sigma_{{\rm th}}={sig_th:.3f}$" + "\n" +

plt.tight_layout(); plt.show()
# CLT 검증 — M_T / sqrt(T) 의 분포 vs N(0, sigma_theory^2)
fig, axes = plt.subplots(len(graphs), 2, figsize=(6, 6.5))
fig.suptitle(r"CLT: $M_T/\sqrt{T} \xrightarrow{d} N(0,\,b^2\rho_i/12)$",
             fontsize=10, y=1.01)
x_grid = np.linspace(-3, 3, 300)

for row, (name, deg) in enumerate(graphs.items()):
    res = results[name]
    for col, (node_idx, node_lbl) in enumerate(node_labels[name]):
        M_node = res["M"][node_idx]
        rho_th = res["prob"][node_idx]
        sig_th = np.sqrt(var_b * rho_th)
        sample = M_node[:, -1] / np.sqrt(T)

        ax = axes[row, col]
        ax.hist(sample, bins=25, density=True, alpha=0.7,
                color=f"C{col}", edgecolor="none")
        ax.plot(x_grid, stats.norm.pdf(x_grid, 0, sig_th),
                lw=1.8, color="crimson",
                label=rf"$N(0,\,{sig_th:.3f}^2)$")
        # KS test
        ks_stat, ks_p = stats.kstest(sample, "norm", args=(0, sig_th))
        ax.set_title(f"{name}: {node_lbl}\n"
                     rf"KS $p={ks_p:.3f}$", fontsize=8)
        ax.set_xlabel(rf"$M_T/\sqrt{{T}}$"); ax.legend(fontsize=7)
        ax.set_xlim(-3, 3)

plt.tight_layout(); plt.show()

KS 검정 p-value 가 모두 0.05 이상이면 \(N(0, b^2\rho_i/12)\) 기각 불가 — 이론치와 일치.

세 그래프 모두에서:

  • \(\hat\rho_i \approx \rho_i^{\text{th}}\) — 장기 점유율이 이론값으로 수렴
  • \(\hat\sigma \approx \sqrt{b^2\rho_i/12}\) — 마팅게일 잔차의 표준편차가 이론 CLT 와 일치
  • KS \(p > 0.05\) — \(N(0, b^2\rho_i/12)\) 분포 기각 불가

Balanced \(C_6\) 에서는 모든 노드의 \(\rho_i = 1/6\) 이 동일해 \(M_t^{(i)}\) 의 분산도 동일. Star 와 Wheel 에서는 hub 의 \(\rho_h > \rho_{\rm leaf}\) 이므로 hub 마팅게일 잔차가 더 크게 흩어진다 — 그래도 \(M_t/t \to 0\) 은 모든 노드에서 보장된다.
